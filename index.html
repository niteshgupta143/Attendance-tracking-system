#[contractimpl]
impl AttendanceContract {
    pub fn mark_attendance(env: Env, student_id: Symbol, session_code: Symbol) -> Result<bool, AttendanceError> {
        if env.storage().persistent().has(&(student_id.clone(), session_code.clone())) {
            return Err(AttendanceError::AlreadyCheckedIn); // Error Type 1
        }
        env.storage().persistent().set(&(student_id.clone(), session_code.clone()), &record);
        env.events().publish((symbol_short!("attend"), session_code), student_id); // Event Integration
        Ok(true)
    }
}
